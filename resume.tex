\section*{\underline{Research Projects}}
\begin{itemize}[leftmargin=*]
    \item \textbf{Physics-Guided Modeling and Simulation}
    \githublink{https://github.com/HaoyangJiang-WM/PhysicsNFP}:
    Developing structure-aware graph operators and physics-guided updates for directional transport, stable spatiotemporal prediction, and long-horizon simulation of networked physical systems.

    \item \textbf{Neural Operators for Boundary Value Problems}:
    \textbf{FIE-NO} \paperlink{https://arxiv.org/abs/2408.12389} uses Fredholm integral equations to learn solution operators under complex and varying boundary conditions.

    \item \textbf{Inverse Modeling with Generative Priors}
    \githublink{https://github.com/HaoyangJiang-WM/FWI}:
    Studying full-waveform inversion with pretrained generative priors and inference-time optimization for reconstruction under new physical configurations.

    \item \textbf{Foundation Models for Forecasting and Data Assimilation}:
    Adapting pretrained spatiotemporal models for observation-guided long-horizon forecasting under sparse data and distribution shift.

    \item \textbf{Second-Order Optimal-Transport Flow Matching}
    \githublink{https://github.com/HaoyangJiang-WM/2otcfm}:
    Developing a phase-space generative framework that couples position, velocity, and acceleration through an optimal-transport variational formulation.

    \item \textbf{Open-System and Boundary-Aware Learning}
    \githublink{https://github.com/HaoyangJiang-WM/Ghost-Node-GNN}:
    Learning latent exterior influence through boundary proxies and physics-based refinement to stabilize prediction in partially observed physical domains.

    \item \textbf{Space Systems and Uncertainty Modeling}:
    Modeled satellite trajectories and studied orbit-error propagation under atmospheric-density uncertainty using machine-learning and numerical methods.
\end{itemize}
